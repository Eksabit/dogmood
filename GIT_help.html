Команды Git

git init  # Создает новый Git-репозиторий в текущей директории
git clone <url>  # Клонирует репозиторий по указанному URL
git status  # Показывает текущее состояние рабочего каталога и индекса
git ls-files # Файлы в индексе
git add <файл>  # Добавляет конкретный файл в индекс для следующего коммита
git add .       # Добавляет все изменения в текущей директории в индекс
git commit -m "Сообщение коммита"  # Сохраняет изменения в репозитории с сообщением
git log  # Показывает список всех коммитов в репозитории
git log --oneline # показывает в одну линию все комиты
git diff              # Показывает изменения в рабочем каталоге по сравнению с последним коммитом
git diff --cached     # Показывает изменения, добавленные в индекс, по сравнению с последним коммитом
git branch <имя-ветки>  # Создает новую ветку с указанным именем
git checkout <имя-ветки>  # Переключается на указанную ветку
git checkout -f <имя-ветки>  # Переключается не сохраняя изменения
git checkout -b <имя-ветки>  # Создает новую ветку и сразу переключается на нее
git merge <имя-ветки>  # Сливает указанную ветку с текущей веткой
git branch -d <имя-ветки>  # Удаляет указанную ветку
git remote add <имя> <url>  # Добавляет удалённый репозиторий с указанным именем и URL
git fetch <имя>  # Получает изменения из удалённого репозитория, но не сливает их
git pull <имя> <ветка>  # Получает и сливает изменения из удалённого репозитория
git push <имя> <ветка>  # Отправляет изменения в указанную ветку удалённого репозитория
git checkout -- <файл>  # Отменяет изменения в указанном файле, возвращая его к последнему коммиту
git reset --hard  # Сбрасывает индекс и рабочий каталог к последнему коммиту, удаляя все несохраненные изменения
git tag <имя-тега>  # Создает новый тег с указанным именем
git tag  # Показывает все теги в репозитории
git add . # Добавляет все файлы в текущей дериктории в репозиторий
git ls-files --cached # Все файлы которые находятся в индексе
